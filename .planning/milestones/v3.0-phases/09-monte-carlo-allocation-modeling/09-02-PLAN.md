5. `run_mc_samples()` function:
   ```python
   def run_mc_samples(
       mc_dist: AllocationDistribution,
       n_samples: int = 50,
       base_seed: int | None = None,
       vote_timing: VoteTimingConfig | None = None,
       *,
       n_voters: int = 200,
       n_proposals: int = 30,
       total_supply: float = 1_000_000.0,
       avg_participation_rate: float = 0.10,
       stake_profile: str = 'pareto',
       lock_profile: str = 'independent',
       pareto_alpha: float = 0.7,
       l_max_days: float = 365.0,
       proposal_window_blocks: int = 50400,
       budget_enabled: bool = True,
       allocation_strategy: str = 'uniform_fraction',
       blocks_per_day: int = 7200,
       curve_type: str = 'sqrt',
   ) -> MCResult:
   ```

   Implementation:
   - If `base_seed is None`, create `ss = np.random.SeedSequence()` and set `base_seed = int(ss.entropy)` if entropy is int, else `int(ss.entropy[0])` (SeedSequence().entropy can be a large int or array — handle both).
     Actually, `np.random.SeedSequence().entropy` returns a single large int when no seed is given. Use `base_seed = ss.entropy`. But store as-is since it can be very large.
   - If `base_seed is not None`, create `ss = np.random.SeedSequence(base_seed)`.
   - `children = ss.spawn(n_samples)` — produces N child SeedSequences.
   - For each child, derive a unique metadata seed: `sample_seed = int(child.generate_state(1)[0])`.
   - Call `generate_scenario()` with `seed=child` (numpy accepts SeedSequence directly as seed to default_rng inside generate_scenario). **IMPORTANT:** Since generate_scenario internally does `np.random.default_rng(seed)` in the mc_dist path via `np.random.SeedSequence(seed)`, we need to pass an int seed instead. Use `sample_seed` as the int seed passed to generate_scenario.

   Actually, re-reading factory.py more carefully: when `mc_dist is not None`, factory.py does `np.random.SeedSequence(seed)` where seed is the int. So we should pass an int. Use `int(child.generate_state(1)[0])` as the seed.

   BUT WAIT — this means two samples could theoretically collide on generate_state output. The research says generate_state produces unique ints from spawn children with distinct spawn_keys. This is safe.

   However, there's a subtlety: `generate_scenario` when `mc_dist is not None` creates ANOTHER SeedSequence from the seed int, then spawns 2 children. This is fine — each sample gets a different int seed, producing different SeedSequences, producing different alloc/scenario RNG streams.

   Loop:
   ```python
   samples = []
   for i, child in enumerate(children):
       sample_seed = int(child.generate_state(1)[0])
       events = generate_scenario(
           n_voters=n_voters,
           n_proposals=n_proposals,
           total_supply=total_supply,
           avg_participation_rate=avg_participation_rate,
           stake_profile=stake_profile,
           lock_profile=lock_profile,
           pareto_alpha=pareto_alpha,
           l_max_days=l_max_days,
           proposal_window_blocks=proposal_window_blocks,
           seed=sample_seed,
           budget_enabled=budget_enabled,
           allocation_strategy=allocation_strategy,
           blocks_per_day=blocks_per_day,
           curve_type=curve_type,
           mc_dist=mc_dist,
           vote_timing=vote_timing,
       )
       # Extract the alloc_frac by re-deriving it from the same seed
       # (since generate_scenario doesn't return it)
       # Instead, compute it directly here for metadata
       alloc_ss = np.random.SeedSequence(sample_seed)
       alloc_child_ss, _ = alloc_ss.spawn(2)
       alloc_rng = np.random.default_rng(alloc_child_ss)
       alloc_frac = float(mc_dist.sample(alloc_rng, 1)[0])

       samples.append(MCSample(
           index=i,
           seed=sample_seed,
           alloc_frac=alloc_frac,
           events=events,
       ))
   ```

   Return `MCResult(base_seed=base_seed, n_samples=n_samples, samples=samples)`.

   ```python
   def test_mcal02_independent_sequences():
       """MCAL-02: 50 MC samples with same base_seed produce 50 distinct allocation sequences."""
       dist = BetaDistribution(a=2, b=5)
       result = run_mc_samples(
           mc_dist=dist,
           n_samples=50,
           base_seed=42,
           n_voters=50,
           n_proposals=3,
       )
       assert result.n_samples == 50
       assert len(result.samples) == 50

       # All allocation fractions must be distinct
       fracs = [s.alloc_frac for s in result.samples]
       unique_fracs = set(round(f, 12) for f in fracs)
       assert len(unique_fracs) == 50, (
           f'Expected 50 distinct alloc fractions, got {len(unique_fracs)}'
       )

       # All fractions must be in [0, 1] (Beta distribution property)
       for f in fracs:
           assert 0.0 <= f <= 1.0, f'alloc_frac {f} out of [0,1] range'
   ```

3. **test_mcal04_mc_runner_distinct_seeds** (MCAL-04):
   ```python
   def test_mcal04_mc_runner_distinct_seeds():
       """MCAL-04: MC runner produces N samples with distinct seeds in metadata."""
       dist = BetaDistribution(a=2, b=5)
       result = run_mc_samples(
           mc_dist=dist,
           n_samples=20,
           base_seed=99,
           n_voters=50,
           n_proposals=3,
       )
       seeds = [s.seed for s in result.samples]
       assert len(set(seeds)) == 20, (
           f'Expected 20 distinct seeds, got {len(set(seeds))}'
       )

       # Each sample has sequential index
       indices = [s.index for s in result.samples]
       assert indices == list(range(20))

       # Each sample has events
       for sample in result.samples:
           assert len(sample.events) > 0, f'Sample {sample.index} has no events'
   ```

4. **test_mc_runner_reproducible** (reproducer from metadata):
   ```python
   def test_mc_runner_reproducible():
       """MC runner with same base_seed produces identical results."""
       dist = BetaDistribution(a=2, b=5)
       r1 = run_mc_samples(mc_dist=dist, n_samples=10, base_seed=42, n_voters=50, n_proposals=3)
       r2 = run_mc_samples(mc_dist=dist, n_samples=10, base_seed=42, n_voters=50, n_proposals=3)

       assert r1.base_seed == r2.base_seed
       for s1, s2 in zip(r1.samples, r2.samples):
           assert s1.seed == s2.seed
           assert s1.alloc_frac == s2.alloc_frac
           assert len(s1.events) == len(s2.events)
   ```
