```python
@dataclass
class VoteTimingConfig:
    """Configuration for vote timing distribution fractions."""
    early: float = 0.30
    mid: float = 0.40
    # late = 1.0 - early - mid (computed, not stored)

    def __post_init__(self):
        if self.early < 0 or self.mid < 0:
            raise ValueError('early and mid must be non-negative')
        if self.early + self.mid > 1.0:
            raise ValueError('early + mid must not exceed 1.0')
```
